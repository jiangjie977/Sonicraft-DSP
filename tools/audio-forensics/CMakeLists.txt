cmake_minimum_required(VERSION 3.22)
project(audio_forensics CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(audio_forensics main.cpp)
target_include_directories(audio_forensics PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../../core/dsp/src/main/cpp)
if(MSVC)
  target_compile_options(audio_forensics PRIVATE /W4 /EHsc /utf-8)
endif()
